    def _read_csv(filepath: str) -> tuple[list[str], list[dict]]:
        # Read as raw bytes for encoding detection
        with open(filepath, "rb") as f:
            raw = f.read()

        # Detect encoding; fall back to utf-8 if chardet is uncertain
        detected = chardet.detect(raw)
        encoding = detected.get("encoding") or "utf-8"
        confidence = detected.get("confidence") or 0.0
        if confidence < 0.5:
            encoding = "utf-8"

        # Decode; fall back to latin-1 which never fails on arbitrary bytes
        try:
            text = raw.decode(encoding, errors="strict")
        except (UnicodeDecodeError, LookupError):
            text = raw.decode("latin-1", errors="replace")

        # Strip UTF-8 BOM if present (equivalent to utf-8-sig behaviour)
        text = text.lstrip("\ufeff")

        lines = text.strip().split("\n")
        # Skip metadata/empty lines — find the header (first line with 2+ commas)
        start = 0
        for i, line in enumerate(lines):
            if line.count(",") >= 2:
                start = i
                break

        reader = csv.DictReader(lines[start:])
        headers = reader.fieldnames or []
        rows = [row for row in reader if any(v.strip() for v in row.values())]
        return headers, rows
    ```

    def _detect_transfer_pairs(account_id: int, imported_tx_ids: list, db: Session) -> int:
        """
        Scan newly imported transactions for potential transfer pairs:
        same absolute amount, within 3 days, across different accounts.
        Returns the count of probable pairs. Does NOT auto-merge (user decision).
        """
        if not imported_tx_ids:
            return 0

        new_txns = db.query(Transaction).filter(Transaction.id.in_(imported_tx_ids)).all()
        if not new_txns:
            return 0

        # Get all other-account transactions in roughly the same date range
        min_date = min(t.date for t in new_txns)
        max_date = max(t.date for t in new_txns)
        from datetime import timedelta
        window_start = min_date - timedelta(days=3)
        window_end = max_date + timedelta(days=3)

        other_txns = db.query(Transaction).filter(
            Transaction.account_id != account_id,
            Transaction.date >= window_start,
            Transaction.date <= window_end,
            Transaction.amount.isnot(None),
        ).all()

        if not other_txns:
            return 0

        pair_count = 0
        for nt in new_txns:
            if nt.amount is None:
                continue
            for ot in other_txns:
                if ot.amount is None:
                    continue
                # Same absolute amount, opposite sign, within 3 days
                if (abs(abs(nt.amount) - abs(ot.amount)) < 0.01 and
                        nt.amount * ot.amount < 0 and
                        abs((nt.date - ot.date).days) <= 3):
                    pair_count += 1
                    break  # count each new transaction once

        return pair_count
    ```
