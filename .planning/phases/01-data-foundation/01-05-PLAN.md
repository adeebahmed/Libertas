    ```typescript
    function staleness(lastUpdated: string | null): 'green' | 'yellow' | 'red' | 'none' {
      if (!lastUpdated) return 'none'
      const days = Math.floor((Date.now() - new Date(lastUpdated).getTime()) / 86400000)
      if (days < 3) return 'green'
      if (days <= 7) return 'yellow'
      return 'red'
    }

        green: 'Updated within 3 days',
        yellow: 'Updated 3-7 days ago',
        red: 'Not updated in over 7 days',
        none: 'Never updated',
      }
      return (
        <span
          title={titles[s]}
          style={{
            display: 'inline-block', width: 8, height: 8, borderRadius: '50%',
            background: colors[s], flexShrink: 0,
    ```

    ```typescript
    function ImportQualityBadges({
      rowsFailed,
      parseErrors,
      potentialTransfers,
    }: {
      rowsFailed?: number
      parseErrors?: string
      potentialTransfers?: number
    }) {
      const [showErrors, setShowErrors] = useState(false)

      const parsedErrors: Array<{ row: Record<string, string>; error: string }> = (() => {
        if (!parseErrors) return []
        try { return JSON.parse(parseErrors) } catch { return [] }
      })()

      return (
        <div style={{ marginTop: 8, display: 'flex', flexDirection: 'column', gap: 6 }}>
          {(rowsFailed ?? 0) > 0 && (
            <div>
              <button
                onClick={() => setShowErrors(v => !v)}
                style={{
                  background: 'none', border: 'none', padding: 0, cursor: 'pointer',
                  color: 'var(--red)', fontSize: 12, display: 'flex', alignItems: 'center', gap: 4,
                }}
              >
                <span>{rowsFailed} row{rowsFailed !== 1 ? 's' : ''} failed to parse</span>
                <span>{showErrors ? '▲' : '▼'}</span>
              </button>
              {showErrors && parsedErrors.length > 0 && (
                <div style={{
                  marginTop: 6, padding: '8px 12px', background: 'var(--bg)',
                  border: '1px solid var(--border)', borderRadius: 6, fontSize: 11,
                  fontFamily: 'monospace', maxHeight: 200, overflowY: 'auto',
                }}>
                  {parsedErrors.map((e, i) => (
                    <div key={i} style={{ marginBottom: 8, borderBottom: i < parsedErrors.length - 1 ? '1px solid var(--border)' : 'none', paddingBottom: 8 }}>
                      <div style={{ color: 'var(--red)', marginBottom: 2 }}>Error: {e.error}</div>
                      <div style={{ color: 'var(--text-3)' }}>Row: {JSON.stringify(e.row).slice(0, 200)}</div>
                    </div>
                  ))}
                </div>
              )}
            </div>
          )}
          {(potentialTransfers ?? 0) > 0 && (
            <div style={{ fontSize: 12, color: 'var(--yellow)' }}>
              {potentialTransfers} potential transfer pair{potentialTransfers !== 1 ? 's' : ''} detected
              {' '}— review manually to avoid double-counting
            </div>
          )}
        </div>
      )
    }
    ```
