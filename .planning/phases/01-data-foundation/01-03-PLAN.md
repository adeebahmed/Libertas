    # --- Manual Entry: Balance ---

    class BalanceSet(BaseModel):
        balance: float
        date: Optional[str] = None  # ISO format YYYY-MM-DD; defaults to today


    @router.post("/{account_id}/balance")
    def set_manual_balance(account_id: int, data: BalanceSet, db: Session = Depends(get_db)):
        """Upsert a balance snapshot for today (or a specified date)."""
        account = db.get(Account, account_id)
        if not account:
            raise HTTPException(404, "Account not found")

        snap_date = date.fromisoformat(data.date) if data.date else date.today()
        existing = db.query(BalanceSnapshot).filter(
            BalanceSnapshot.account_id == account_id,
            BalanceSnapshot.date == snap_date,
        ).first()
        if existing:
            existing.balance = data.balance
        else:
            db.add(BalanceSnapshot(account_id=account_id, date=snap_date, balance=data.balance))
        db.commit()
        return {"ok": True, "date": snap_date.isoformat(), "balance": data.balance}


    # --- Manual Entry: Transactions ---

    class TransactionCreate(BaseModel):
        date: str                         # ISO format YYYY-MM-DD
        type: str                         # buy | sell | other
        amount: Optional[float] = None
        description: Optional[str] = None
        symbol: Optional[str] = None
        quantity: Optional[float] = None
        price: Optional[float] = None


    @router.post("/{account_id}/transactions")
    def create_manual_transaction(account_id: int, data: TransactionCreate, db: Session = Depends(get_db)):
        """Create a manual transaction (no CSV import_hash)."""
        account = db.get(Account, account_id)
        if not account:
            raise HTTPException(404, "Account not found")

        try:
            tx_date = date.fromisoformat(data.date)
        except ValueError:
            raise HTTPException(400, f"Invalid date format: {data.date!r} — expected YYYY-MM-DD")

        tx = Transaction(
            account_id=account_id,
            import_log_id=None,
            date=tx_date,
            type=data.type,
            symbol=data.symbol,
            quantity=data.quantity,
            price=data.price,
            amount=data.amount,
            description=data.description,
            raw_row=None,
            import_hash=None,   # None marks manual entry
        )
        db.add(tx)
        db.commit()
        db.refresh(tx)
        return {
            "id": tx.id,
            "date": tx.date.isoformat(),
            "type": tx.type,
            "symbol": tx.symbol,
            "quantity": tx.quantity,
            "price": tx.price,
            "amount": tx.amount,
            "description": tx.description,
            "import_log_id": tx.import_log_id,
        }


    @router.delete("/{account_id}/transactions/{tx_id}")
    def delete_manual_transaction(account_id: int, tx_id: int, db: Session = Depends(get_db)):
        """Delete a manually entered transaction. Refuses to delete CSV-imported transactions."""
        tx = db.get(Transaction, tx_id)
        if not tx or tx.account_id != account_id:
            raise HTTPException(404, "Transaction not found")
        if tx.import_hash is not None:
            raise HTTPException(403, "Cannot delete CSV-imported transactions; use import rollback instead")
        db.delete(tx)
        db.commit()
        return {"ok": True}


    # --- Manual Entry: Holdings ---

    class HoldingCreate(BaseModel):
        symbol: str
        quantity: float
        cost_basis: Optional[float] = None
        last_price: Optional[float] = None


    class HoldingUpdate(BaseModel):
        symbol: Optional[str] = None
        quantity: Optional[float] = None
        cost_basis: Optional[float] = None
        last_price: Optional[float] = None


    @router.post("/{account_id}/holdings")
    def create_manual_holding(account_id: int, data: HoldingCreate, db: Session = Depends(get_db)):
        """Create a manual holding position (source='manual', survives CSV rebuild)."""
        account = db.get(Account, account_id)
        if not account:
            raise HTTPException(404, "Account not found")

        holding = Holding(
            account_id=account_id,
            symbol=data.symbol.upper().strip(),
            quantity=data.quantity,
            cost_basis=data.cost_basis,
            last_price=data.last_price,
            source="manual",
        )
        db.add(holding)
        db.commit()
        db.refresh(holding)
        return {
            "id": holding.id,
            "symbol": holding.symbol,
            "quantity": holding.quantity,
            "cost_basis": holding.cost_basis,
            "last_price": holding.last_price,
            "source": holding.source,
        }


    @router.patch("/{account_id}/holdings/{holding_id}")
    def update_holding(account_id: int, holding_id: int, data: HoldingUpdate, db: Session = Depends(get_db)):
        """Update a holding's symbol, quantity, cost_basis, or last_price."""
        holding = db.get(Holding, holding_id)
        if not holding or holding.account_id != account_id:
            raise HTTPException(404, "Holding not found")
        for k, v in data.model_dump(exclude_unset=True).items():
            if k == "symbol" and v:
                v = v.upper().strip()
            setattr(holding, k, v)
        db.commit()
        db.refresh(holding)
        return {
            "id": holding.id,
            "symbol": holding.symbol,
            "quantity": holding.quantity,
            "cost_basis": holding.cost_basis,
            "last_price": holding.last_price,
            "source": holding.source,
        }


    @router.delete("/{account_id}/holdings/{holding_id}")
    def delete_holding(account_id: int, holding_id: int, db: Session = Depends(get_db)):
        """Delete a holding."""
        holding = db.get(Holding, holding_id)
        if not holding or holding.account_id != account_id:
            raise HTTPException(404, "Holding not found")
        db.delete(holding)
        db.commit()
        return {"ok": True}
